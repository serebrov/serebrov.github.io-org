#+TITLE: git: count commits by author
#+DATE: <2021-09-13>
#+TYPE: note
#+FILETAGS: :git:
#+DESCRIPTION: Git: How to show the number of commits by author

The =shortlog -ns= will show the number of commits by author:

#+begin_src shell
git shortlog -ns
#+end_src

#+begin_example
   280  Author One
    46  Author Two
    25  authorthree
    14  au
     4  x
     3  Autor One
     1  ide user x
#+end_example
